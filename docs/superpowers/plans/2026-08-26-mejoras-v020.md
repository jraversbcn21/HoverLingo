# HoverLingo v0.2.0 Improvements Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the 4 pending Future Work improvements (SW request collapsing, popup i18n EN+ES, stats reset button, SW integration tests) and release v0.2.0 to the Chrome Web Store.

**Architecture:** All server-side logic lives in the MV3 service worker (`src/background/service-worker.ts`); the popup reads/writes `chrome.storage.local` directly. Request collapsing adds an in-memory pending-promise Map to the SW keyed by the existing L2 cache key. i18n uses native `chrome.i18n` with `_locales` under `public/`. Tests use vitest with stubbed `chrome` and `fetch` globals, following the pattern in `src/__tests__/cache-l2.test.ts`.

**Tech Stack:** TypeScript 5 (strict), Vite 6 + @crxjs/vite-plugin, vitest 3, Chrome MV3 APIs (`chrome.storage`, `chrome.i18n`, `chrome.runtime`).

```typescript
import type { TranslationResponse } from "../shared/types";
import { buildCacheKey } from "../shared/cache-key";

const mockStore = new Map<string, unknown>();

type Listener = (
  message: { type: string; payload?: unknown },
  sender: unknown,
  sendResponse: (response: unknown) => void
) => boolean;

let listener: Listener;

vi.stubGlobal("chrome", {
  storage: {
    local: {
      get: vi.fn((keys: string | string[] | null) => {
        const result: Record<string, unknown> = {};
        if (keys === null) {
          for (const [k, v] of mockStore) result[k] = v;
        } else if (Array.isArray(keys)) {
          for (const k of keys) {
            if (mockStore.has(k)) result[k] = mockStore.get(k);
          }
        } else if (typeof keys === "string") {
          if (mockStore.has(keys)) result[keys] = mockStore.get(keys);
        }
        return Promise.resolve(result);
      }),
      set: vi.fn((items: Record<string, unknown>) => {
        for (const [k, v] of Object.entries(items)) mockStore.set(k, v);
        return Promise.resolve();
      }),
      remove: vi.fn((keys: string | string[]) => {
        for (const k of Array.isArray(keys) ? keys : [keys]) mockStore.delete(k);
        return Promise.resolve();
      }),
    },
  },
  runtime: {
    onMessage: {
      addListener: vi.fn((fn: Listener) => {
        listener = fn;
      }),
    },
  },
});

const fetchMock = vi.fn();
vi.stubGlobal("fetch", fetchMock);

interface SWResponse {
  success: boolean;
  data?: TranslationResponse;
  error?: string;
  cached?: boolean;
}

function sendTranslate(payload: {
  text: string;
  sentence: string;
  targetLang: string;
  mode: "quick" | "learning";
}): Promise<SWResponse> {
  return new Promise((resolve) => {
    listener({ type: "TRANSLATE", payload }, {}, (r) => resolve(r as SWResponse));
  });
}

function groqResponse(json: object): Response {
  return new Response(
    JSON.stringify({ choices: [{ message: { content: JSON.stringify(json) } }] }),
    { status: 200 }
  );
}

const TRANSLATION = {
  translation: "casa",
  sourceLanguage: "en",
  direction: "ltr",
  confidence: 0.95,
};

const PAYLOAD = {
  text: "house",
  sentence: "The house is big.",
  targetLang: "es",
  mode: "quick" as const,
};

const DEFAULT_MODEL = "qwen/qwen3.6-27b";

beforeEach(async () => {
  vi.resetModules();
  mockStore.clear();
  fetchMock.mockReset();
  mockStore.set("groqApiKey", "gsk_test");
  await import("../background/service-worker");
});

describe("Service worker integration", () => {
  it("answers from L2 cache without calling fetch", async () => {
    const key = buildCacheKey(PAYLOAD.text, PAYLOAD.sentence, "es", "quick", DEFAULT_MODEL);
    mockStore.set(`hl_cache_${key}`, { response: TRANSLATION, timestamp: Date.now() });

    const res = await sendTranslate(PAYLOAD);

    expect(res.success).toBe(true);
    expect(res.cached).toBe(true);
    expect(res.data).toEqual(TRANSLATION);
    expect(fetchMock).not.toHaveBeenCalled();
  });

  it("on L2 miss calls Groq once, persists to L2 and answers cached:false", async () => {
    fetchMock.mockResolvedValue(groqResponse(TRANSLATION));

    const res = await sendTranslate(PAYLOAD);

    expect(res.success).toBe(true);
    expect(res.cached).toBe(false);
    expect(res.data?.translation).toBe("casa");
    expect(fetchMock).toHaveBeenCalledTimes(1);

    const key = buildCacheKey(PAYLOAD.text, PAYLOAD.sentence, "es", "quick", DEFAULT_MODEL);
    const stored = mockStore.get(`hl_cache_${key}`) as { response: TranslationResponse };
    expect(stored.response.translation).toBe("casa");
  });

  it("fails cleanly when no API key is configured", async () => {
    mockStore.delete("groqApiKey");

    const res = await sendTranslate(PAYLOAD);

    expect(res.success).toBe(false);
    expect(res.error).toBe("API key not configured");
    expect(fetchMock).not.toHaveBeenCalled();
  });

  it("maps persistent 429 to RATE_LIMIT error", async () => {
    fetchMock.mockResolvedValue(
      new Response('{"error":{"message":"rate limit"}}', {
        status: 429,
        headers: { "Retry-After": "0" },
      })
    );
    // El SW reintenta una vez el 429 (con Retry-After 0 la espera es
    // instantánea); ambos intentos devuelven 429 → error RATE_LIMIT.
    const res = await sendTranslate(PAYLOAD);

    expect(res.success).toBe(false);
    expect(res.error).toBe("RATE_LIMIT");
    expect(fetchMock).toHaveBeenCalledTimes(2);
  });

  it("falls back to default model when stored model is decommissioned", async () => {
    mockStore.set("groqModel", "llama-3.3-70b-versatile");
    fetchMock.mockResolvedValue(groqResponse(TRANSLATION));

    const res = await sendTranslate(PAYLOAD);

    expect(res.success).toBe(true);
    const body = JSON.parse(fetchMock.mock.calls[0][1].body as string);
    expect(body.model).toBe(DEFAULT_MODEL);
    expect(body.reasoning_effort).toBe("none");
  });
});
```

```typescript
    const existing = inFlight.get(cacheKey);
    if (existing) {
      const data = await existing;
      sendResponse({ success: true, data, cached: true });
      return;
    }

    const flight = (async () => {
      const result = await callGroqWithRetry(
        request.text,
        request.sentence,
        targetLang,
        mode,
        apiKey,

      );

      if (!validateTranslationScript(result.translation, targetLang)) {
        result.confidence = 0;
      }

      await l2Cache.set(cacheKey, result);
      return result;
    })();
    inFlight.set(cacheKey, flight);

      sendResponse({ success: true, data: result, cached: false });
    } finally {
      inFlight.delete(cacheKey);
    }
```

Notes for the implementer:
- The outer `try/catch` in `handleTranslate` already converts rejections (including joiners' `await existing`) into error responses, with the 429→`RATE_LIMIT` special case. Don't duplicate error handling.
- `inFlight.set` runs synchronously right after creating the promise — no `await` between creation and registration, or a concurrent request could slip past the check.

```typescript
function msg(key: string, ...subs: string[]): string {

function localize(): void {

}
```
